\section{Limitations}
\label{sec:limitations}

\subsection{Current Limitations}
\begin{enumerate}
    \item \textbf{Testnet Only:} Not production-hardened; Mainnet requires governance review.
    \item \textbf{Pseudonymous, Not Anonymous:} Voter ID linked to vote in DB (though not on-chain).
    \item \textbf{No Coercion Resistance:} Voters can prove how they voted via transaction ID.
    \item \textbf{Single Election:} One active election per deployment; multi-election needs dynamic topics.
    \item \textbf{Manual Key Management:} Operator keys in .env; production needs HSM/KMS.
    \item \textbf{No Vote Privacy:} Mirror nodes reveal vote content; ZK-proofs needed for secrecy.
    \item \textbf{Client-Side Trust:} Malicious frontend could alter vote before submission.
\end{enumerate}

\begin{itemize}
    \item Multi-election support with dynamic HCS topic creation
    \item Automated CI/CD pipeline with testnet integration tests
    \item Enhanced monitoring (Prometheus/Grafana dashboards)
    \item Accessibility audit (WCAG 2.1 AA compliance)
    \item Load testing with Hedera Previewnet (higher throughput)

\subsection{Medium Term (6--12 months)}
\begin{itemize}
    \item Zero-knowledge proof integration for vote privacy (zk-SNARKs)
    \item Receipt-freeness via blind signatures or mixnets
    \item Mainnet deployment with Hedera Council governance review
    \item Mobile app (React Native) with biometric auth option
    \item Vote delegation (liquid democracy) smart contracts
\end{itemize}

\subsection{Long Term (12+ months)}
\begin{itemize}
    \item Post-quantum cryptography migration (CRYSTALS-Dilithium)
    \item Formal verification of vote atomicity (TLA+)
    \item Cross-chain interoperability for federated elections
    \item AI-assisted manifesto analysis for voters
    \item Integration with national digital identity frameworks
\end{itemize}

\section{Ethical and Societal Considerations}
\label{sec:ethical}

\begin{itemize}
    \item \textbf{Inclusivity:} Web-based access excludes offline populations; hybrid paper-digital needed.
    \item \textbf{Transparency vs Privacy:} Public auditability conflicts with ballot secrecy; ZK-proofs can reconcile.
    \item \textbf{Trust in Technology:} Non-technical voters must trust cryptographic guarantees; education essential.
    \item \textbf{Governance:} Who controls the HCS topic? Multi-sig admin keys recommended.
    \item \textbf{Dispute Resolution:} On-chain finality means no rollback; procedural safeguards required.
\end{itemize}
