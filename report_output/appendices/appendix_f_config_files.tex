\section{Docker Compose}
\label{sec:docker-compose}
\begin{lstlisting}[style=codeStyle, caption=docker-compose.yml]
version: '3.8'

services:
  mongodb:
    image: mongo:8
    container_name: evoting-mongodb
    volumes:
      - mongodb_data:/data/db
    ports:
      - "27017:27017"
    environment:
      MONGO_INITDB_DATABASE: evoting_db

  backend:
    build: ./backend
    container_name: evoting-backend
    ports:
      - "5000:5000"
    environment:
      - MONGO_URI=mongodb://mongodb:27017/evoting_db
      - JWT_SECRET=${JWT_SECRET}
      - HEDERA_NETWORK=${HEDERA_NETWORK}
      - HEDERA_ACCOUNT_ID=${HEDERA_ACCOUNT_ID}
      - HEDERA_PRIVATE_KEY=${HEDERA_PRIVATE_KEY}
      - HCS_TOPIC_ID=${HCS_TOPIC_ID}
    depends_on:
      - mongodb
    env_file:
      - backend/.env

  frontend:
    build: ./frontend
    container_name: evoting-frontend
    ports:
      - "5173:80"
    depends_on:
      - backend

volumes:
  mongodb_data:
\end{lstlisting}

import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'

export default defineConfig({
  plugins: [react()],
  server: {
    port: 5173,
    proxy: {
      '/api': {
        target: 'http://localhost:5000',
        changeOrigin: true
      }
    }
  },
  build: {
    outDir: 'dist',
    sourcemap: true
  }
})
\end{lstlisting}

\section{Nginx Configuration}
\label{sec:nginx-config}
\begin{lstlisting}[style=codeStyle, caption=frontend/nginx.conf]
server {
    listen 80;
    server_name localhost;
    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    location /api {
        proxy_pass http://backend:5000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}
\end{lstlisting}

\section{Backend Dockerfile}
\label{sec:backend-dockerfile}
\begin{lstlisting}[style=codeStyle, caption=backend/Dockerfile]
FROM node:22-alpine
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production
COPY . .
EXPOSE 5000
CMD ["node", "server.js"]
\end{lstlisting}

\section{Frontend Dockerfile}
\label{sec:frontend-dockerfile}
\begin{lstlisting}[style=codeStyle, caption=frontend/Dockerfile]
FROM node:22-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build

FROM nginx:alpine
COPY --from=builder /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
\end{lstlisting}
