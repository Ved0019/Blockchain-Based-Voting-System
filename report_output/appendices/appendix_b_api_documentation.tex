\section{Base URL}
\label{sec:base-url}
Development: \texttt{http://localhost:5000/api} \\
Production: Not configured in the current repository

\section{Authentication}
\label{sec:api-auth}
All protected endpoints require \texttt{Authorization: Bearer <JWT>} header.
JWT obtained via \texttt{POST /api/auth/login} with \texttt{\{voterId, password\}}.

\section{Request/Response Examples}
\label{sec:api-examples}

\subsection{Login}
\begin{lstlisting}[style=codeStyle, caption=Login request/response]
POST /api/auth/login
{ "voterId": "VOTER001", "password": "voter123" }

Response 200:
{ "message": "Login successful", "token": "eyJhbGciOiJIUzI1NiIs...",
  "user": { "voterId": "VOTER001", "role": "Voters" } }
\end{lstlisting}

\subsection{Vote Submission}
\begin{lstlisting}[style=codeStyle, caption=Vote submission request/response]
POST /castVote
Authorization: Bearer <token>
{ "candidateId": "1" }

Response 200:
{ "message": "Vote recorded on Hedera!",
  "transactionId": "0.0.12345@1234567890.123456789",
  "status": "SUCCESS" }
\end{lstlisting}

\subsection{Error Responses}
\begin{lstlisting}[style=codeStyle, caption=Common error responses]
401: { "error": "Access token required" }
403: { "error": "Invalid or expired token" }
403: { "error": "This voter has already voted." }
400: { "error": "Invalid candidate selected." }
502: { "error": "Ledger submission failed, vote not recorded" }
500: { "error": "Failed to broadcast vote to ledger." }
\end{lstlisting}
