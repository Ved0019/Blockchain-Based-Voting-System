\section{Key File Descriptions}
\label{sec:key-files}

\subsection{Backend}
\begin{table}[H]
\centering
\caption{Key Backend Files}
\label{tab:backend-files}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{File} & \textbf{Purpose} \\
\midrule
server.js & Express app entry, middleware, routes, Hedera init \\
controllers/authController.js & Login, JWT issuance \\
controllers/electionController.js & Election/candidate CRUD \\
controllers/voteController.js & Vote casting, verification \\
models/User.js & Voter schema (voterId, name, passwordHash, role, hasVoted) \\
models/Election.js & Election schema (title, dates, status, topicId) \\
models/Candidate.js & Candidate schema (manifesto[], performance[]) \\
models/CampusIssue.js & Issue schema (title, category, urgency, solutions[]) \\
models/EvidenceDoc.js & Evidence schema (docId, title, type, status) \\
lib/hederaClient.js & Hedera SDK client, HCS submission \\
seedData.js & Demo data: 4 candidates, 6 issues, 8 evidence docs \\
createTopic.js & HCS topic creation script \\
\midrule
\end{tabular}
\end{table}
